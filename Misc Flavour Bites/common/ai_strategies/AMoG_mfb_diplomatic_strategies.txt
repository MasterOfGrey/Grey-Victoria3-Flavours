# Unification

TRY_REPLACE:ai_strategy_unify_germany = {
	icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = yes
	}

	undesirable_infamy_level = {
		value = 50
	}

	unacceptable_infamy_level = {
		value = 100
	}

	ideological_opinion_effect_mult = {
		value = 0.5
	}

	wargoal_maneuvers_fraction = {
		value = 0.2
	}

	diplomatic_play_neutrality = {
		value = 0
	}

	diplomatic_play_boldness = {
		value = 75
	}

	obligation_value = {
		value = 0

		if = {
			limit = {
				scope:target_country = {
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
				}
			}
			add = 25
		}
	}

	recklessness = {
		value = 0.5

		if = {
			limit = {
				has_technology_researched = enlistment_offices
			}
			add = 0.25
		}
	}

	aggression = {
		value = -2.0

		if = {
			limit = {
				scope:target_country = {
					is_unification_candidate = GER
				}
				NOT = { has_journal_entry = je_schleswig_holstein_question }
			}
			add = 0.25

			if = {
				limit = {
					has_technology_researched = enlistment_offices
				}
				add = 10
			}
		}
		else_if = {
			limit = {
				scope:target_country = {
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
						NOT = { is_unification_candidate = GER }
					}
				}
			}
			add = 0.25

			if = {
				limit = {
					has_technology_researched = enlistment_offices
				}
				add = 2
			}
		}

		# Schleswig-Holstein question
		if = {
			limit = {
				wants_to_solve_schleswig_holstein = yes
				scope:target_country = {
					is_oppressor_of_schleswig_holstein = yes
				}
			}
			add = 15
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.25
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}
	}

	building_group_weights = {
		bg_army = 1.5
		bg_military_industry = 1.5
	}

	state_value = {
		if = {
			limit = {
				scope:target_state = {
					state_region = {
						OR = {
							is_homeland = cu:north_german
							is_homeland = cu:south_german
						}
					}
				}
				can_reach_target_state = yes
			}
			add = 100
		}
	}

	wargoal_scores = {
		unification_leadership = {
			value = 5
			if = { # Fervour
				limit = {
					has_technology_researched = pan-nationalism
				}
				add = {
					value = country_fervor_primary_culture
					divide = 2
				}
			}
			else = {
				add = {
					value = country_fervor_primary_culture
					divide = 10
				}
			}
			if = { # At this point - up to 55 if Pan-National, 15 if not
				limit = {
					any_political_movement ?= {
						is_political_movement_type = movement_german_national
						save_temporary_scope_as = german_national_movement
					}
				}
				scope:german_national_movement ?= {
					multiply = {
						value = 1
						add = this.political_movement_radicalism # At this point - maximum 110 [if Pan-National] to 30 [if not]
					}
				}
				if = {
					limit = {
						scope:german_national_movement ?= {
							any_influenced_interest_group = {
								is_in_government = yes
								is_powerful = no
							}
						}
					}
					multiply = 1.5 # Maximum 155 if Pan-National, 45 if not
				}
				else_if = {
					limit = {
						scope:german_national_movement ?= {
							any_influenced_interest_group = {
								is_in_government = yes
								is_powerful = yes
							}
						}
					}
					multiply = 2.5 # Maximum 275 if Pan-National, 75 if not
				}
			}
			if = {
				limit = {
					has_law = law_type:law_subjecthood
				}
				multiply = 0.1
			}
		}
	}

	secret_goal_scores = {
		protect = {
			trigger = {
				scope:target_country = {
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
					NOT = { is_unification_candidate = GER }
					OR = {
						is_supporting_unification_candidate = { who = root country_formation = GER }
						is_subject_of = root
						is_in_customs_union_with = root
					}
				}
			}
			score = {
				value = 1000
			}
		}
		dominate = {
			trigger = {
				scope:target_country = {
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
					NOT = { is_unification_candidate = GER }
					NOR = {
						is_supporting_unification_candidate = { who = root country_formation = GER }
						is_subject_of = root
						is_in_customs_union_with = root
					}
				}
			}
			score = {
				value = 1000
			}
		}
		conquer = {
			trigger = {
				scope:target_country = {
					is_unification_candidate = GER
				}
			}
			score = {
				value = 1000
			}
		}
	}

	possible = {
		is_subject = no
		OR = {
			AND = {
				is_unification_candidate = GER
				NOT = { country_tier = empire }
			}
			has_journal_entry = je_german_unification		# MFB
		}
	}

	weight = {
		value = 5
		if = { # Fervour
			limit = {
				has_technology_researched = pan-nationalism
			}
			add = {
				value = country_fervor_primary_culture
				divide = 2
			}
		}
		else = {
			add = {
				value = country_fervor_primary_culture
				divide = 10
			}
		}
		if = { # At this point - up to 55 if Pan-National, 15 if not
			limit = {
				any_political_movement ?= {
					is_political_movement_type = movement_german_national
					save_temporary_scope_as = german_national_movement
				}
			}
			scope:german_national_movement ?= {
				multiply = {
					value = 1
					add = this.political_movement_radicalism # At this point - maximum 110 [if Pan-National] to 30 [if not]
				}
			}
			if = {
				limit = {
					scope:german_national_movement ?= {
						any_influenced_interest_group = {
							is_in_government = yes
							is_powerful = no
						}
					}
				}
				multiply = 1.5 # Maximum 155 if Pan-National, 45 if not
			}
			else_if = {
				limit = {
					scope:german_national_movement ?= {
						any_influenced_interest_group = {
							is_in_government = yes
							is_powerful = yes
						}
					}
				}
				multiply = 2.5 # Maximum 275 if Pan-National, 75 if not
			}
		}
		if = {
			limit = {
				has_law = law_type:law_subjecthood
			}
			multiply = 0.1
		}
	}
}
