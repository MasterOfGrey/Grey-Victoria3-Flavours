## Mod Deconflictions exist to prevent the AI from forming the "wrong" nation with other random formables when there's special content for one in this mod and the others are "just" formable options.

# Kikko's formables

TRY_REPLACE:ki_ZAP = { #Zapadoslavia
	# use_culture_states = yes
	states = {
		STATE_TAHITI
	}

	required_states_fraction = 1
	
	ai_will_do = { always = no }

	possible = {
		NOT = { Grey_MFB_is_active = yes }
	}
}
