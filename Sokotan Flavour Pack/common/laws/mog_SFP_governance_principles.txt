## Islamic theocracy variant.
law_sfp_caliphate = {
	group = lawgroup_governance_principles
	parent = law_theocracy

	icon = "gfx/interface/icons/law_icons/theocracy.dds"

	progressiveness = -25

	is_visible = {
		mog_valid_caliphate = yes
	}

	disallowing_laws = {
		law_total_separation
	}

	can_enact = {
		trigger_if ={
			limit = {
				is_subject = no
			}
			NOT = { has_government_type = gov_chartered_company }
		}
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_governance_principles_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	on_activate = {
		if = {
			limit = {
				has_law = law_type:law_neo_absolutism
			}
			deactivate_law = law_type:law_neo_absolutism
			activate_law = law_type:law_autocracy
		}
	}

	on_deactivate = {
		if = {
			limit = {
				AND = {
					has_law = law_type:law_subjecthood
					NOR = {
						has_law = law_type:law_monarchy
						is_enacting_law = law_type:law_monarchy
					}
				}
			}
			if = {
				limit = {
					any_interest_group = {
						is_in_government = yes
						any_preferred_law = {
							is_citizenship_law = yes
							NOT = {
								type ?= law_type:law_subjecthood
							}
						}
					}
				}
				ordered_interest_group = {
					limit = {
						is_in_government = yes
						any_preferred_law = {
							is_citizenship_law = yes
							NOT = {
								type ?= law_type:law_subjecthood
							}
						}
					}
					order_by = ig_clout
					position = 0
					check_range_bounds = no
					ordered_preferred_law = {
						limit = {
							is_citizenship_law = yes
							NOT = {
								type ?= law_type:law_subjecthood
							}
						}
						position = 0
						check_range_bounds = no
						prev.owner = {
							activate_law = prev.type
						}
					}
				}
			}
			else = {
				activate_law = law_type:law_national_supremacy
			}
		}
	}

	modifier = {
		country_legitimacy_headofstate_add = 20
		country_legitimacy_ideological_incoherence_mult = 0.1
		interest_group_ig_devout_pol_str_mult = 0.25
		country_authority_add = 200
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			has_ideology = ideology:ideology_theocrat
			has_religion = sunni
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = {
				AND = {
					ai_has_enact_weight_modifier_journal_entries = yes
					mog_valid_caliphate = yes
				}
			}
			add = 750
		}
	}

	ai_impose_chance = {
		value = 0
	}
}

		mog_valid_caliphate = no
	}
}
