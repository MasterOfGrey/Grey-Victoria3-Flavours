### The Theocracy replacement caliphate.
REPLACE:gov_caliphate = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_CALIPH"
	female_ruler = "" # Empty on purpose

	male_heir = "RULER_TITLE_CALIPHPRINCE"
	female_heir = "" # Empty on purpose

	possible = {
		has_law = law_type:law_sfp_caliphate
		has_gov_regency = no
	}

	on_government_type_change = {
		change_to_hereditary = yes
	}
	on_post_government_type_change = {
		post_change_to_hereditary = yes
	}
}

## Overwrite to avoid a clash:
REPLACE:gov_sunni_theocracy = {
	transfer_of_power = dictatorial
	male_ruler = "RULER_IMAM"
	female_ruler = "RULER_IMAM"

	possible = {
		NOT = { has_law = law_type:law_sfp_caliphate }
		has_law_or_variant = law_type:law_theocracy
		OR = {
			country_has_state_religion = rel:sunni
			country_has_state_religion = rel:ibadi
		}
	}

	on_government_type_change = {
		change_to_dictatorial = yes
	}
	on_post_government_type_change = {
		post_change_to_dictatorial = yes
	}
}
